# EMI MPU v2 (32 региона / 16 доменов)
# repo github.com/nokia-dev/android_kernel_nokia_mt6771 (HEAD)
# path drivers/misc/mediatek/emi_mpu/mt6758/emi_mpu.h
# NB: платформенной папки mt6771 в этом дереве НЕТ (есть mt6739/mt6757/mt6758);
#     вариант сверен по числу регионов/доменов с тем, что использует вендор MT6771 (регионы 25/26, 16 доменов).

#define EMI_MPU_CTRL                 (0x0)
#define EMI_MPU_DBG                  (0x4)
#define EMI_MPU_SA0                  (0x100)
#define EMI_MPU_EA0                  (0x200)
#define EMI_MPU_SA(region)           (EMI_MPU_SA0 + (region*4))
#define EMI_MPU_EA(region)           (EMI_MPU_EA0 + (region*4))
#define EMI_MPU_APC0                 (0x300)
#define EMI_MPU_APC(region, domain)  (EMI_MPU_APC0 + (region*4) + ((domain/8)*0x100))
#define EMI_MPU_CTRL_D0              (0x800)
#define EMI_MPU_CTRL_D(domain)       (EMI_MPU_CTRL_D0 + (domain*4))

#define EMI_MPU_REGION_NUMBER	(32)
#define EMI_MPU_DOMAIN_NUMBER	(16)
