# Cross-file для glmark2 (C++!) — отличается от tools/cross-aarch64-mesa.txt.
#
# Почему без -isystem на sysroot: glmark2 — C++, и <cmath> из кросс-g++
# (gcc-cross 15) тянет <math.h> через `#include_next`, который продолжает
# поиск ПОСЛЕ каталога, где найден cmath, то есть в /usr/aarch64-linux-gnu/
# include. Если C-заголовки sysroot стоят в -isystem (раньше каталогов
# тулчейна), получается math.h от хостовой glibc + bits/mathcalls.h от
# sysroot → «__intmax_t does not name a type». Поэтому C/C++-заголовки берём
# комплектом от тулчейна, а sysroot используем только для библиотек
# (--sysroot) и pkg-config (EGL/GLESv2/GBM/...). Для программы, которая
# линкуется с libc планшета, это безопасно: новые символы glibc отсекаются
# на этапе линковки.
[binaries]
c = 'aarch64-linux-gnu-gcc'
cpp = 'aarch64-linux-gnu-g++'
ar = 'aarch64-linux-gnu-ar'
strip = 'aarch64-linux-gnu-strip'
pkg-config = 'pkg-config'

[properties]
sys_root = '/home/l/t1100-sysroot'
pkg_config_libdir = ['/home/l/t1100-sysroot/usr/lib/aarch64-linux-gnu/pkgconfig',
                     '/home/l/t1100-sysroot/usr/lib/pkgconfig',
                     '/home/l/t1100-sysroot/usr/share/pkgconfig']

[built-in options]
c_args = ['--sysroot=/home/l/t1100-sysroot']
cpp_args = ['--sysroot=/home/l/t1100-sysroot']
c_link_args = ['--sysroot=/home/l/t1100-sysroot',
               '-L/home/l/t1100-sysroot/usr/lib/aarch64-linux-gnu',
               '-L/home/l/t1100-sysroot/lib/aarch64-linux-gnu']
cpp_link_args = ['--sysroot=/home/l/t1100-sysroot',
                 '-L/home/l/t1100-sysroot/usr/lib/aarch64-linux-gnu',
                 '-L/home/l/t1100-sysroot/lib/aarch64-linux-gnu']

[host_machine]
system = 'linux'
cpu_family = 'aarch64'
cpu = 'aarch64'
endian = 'little'
